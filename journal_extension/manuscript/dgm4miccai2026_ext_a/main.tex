\begin{abstract}
Scanner- and site-induced intensity shift confounds multi-institutional MRI
studies. Unpaired image translation addresses this through two largely
disjoint content-preservation mechanisms: global, bidirectional
cycle-consistency, and local, feature-space patch-contrastive alignment
(PatchNCE). We argue these mechanisms fail in complementary ways under
unpaired glioma data --- the cycle constraint admits an appearance-leaking,
boundary-smoothing equilibrium, while one-sided PatchNCE lacks a global
appearance anchor --- and we study their combination on a self-attention
2.5D tri-planar backbone. We frame the work as a mechanism study: rather than
asking ``does adding PatchNCE help,'' we ask under what weighting, on what
backbone, and measured by which downstream task the cycle/contrastive
combination resolves a failure mode that neither component resolves alone.
On a BraTS\,$\leftrightarrow$\,UPenn-GBM benchmark we evaluate with
distribution-matching metrics (\todo{FID/KID, domain-classifier accuracy}),
brain-masked structural similarity, and \emph{downstream} cross-site glioma
sub-region segmentation (\todo{Dice/HD95}), and we ablate the contrastive
weight $\lambda_{\mathrm{NCE}}\in\{0.1,0.5,1.0,2.0\}$.

\end{abstract}

\section{Related Work}
\label{sec:related}
\paragraph{Contrastive unpaired translation.}
CUT~\cite{park2020cut} replaced cycle-consistency with a one-sided multi-layer
PatchNCE objective; DCLGAN~\cite{han2021dclgan} showed contrastive and
cycle-style dual objectives are complementary on natural images; subsequent
work refined patch sampling and negatives (QS-Attn~\cite{hu2022qsattn},
SRC~\cite{jung2022src}, NEGCUT~\cite{wang2021negcut}, MoNCE~\cite{zhan2022monce}),
all rooted in InfoNCE~\cite{oord2018cpc}. \todo{1--2 sentences positioning our
\emph{re-introduction} of full cycle on top of PatchNCE as the inverse design
choice we justify, not hide.}

\paragraph{MRI harmonization.}
ComBat~\cite{fortin2017combat} corrects batch effects in \emph{features}, not
images, and cannot serve spatial downstream tasks. Disentanglement methods
(CALAMITI~\cite{zuo2021calamiti}, HACA3~\cite{zuo2023haca3}) and the 2.5D
VAE-GAN ImUnity~\cite{cackowski2023imunity} are the closest image-space
neighbors; IGUANe~\cite{roca2024iguane} is a strong pure-cycle baseline. The
closest methodological precedent, cycleCUT~\cite{wang2022cyclecut}, combines
cycle and contrastive losses but for cross-modality MR$\to$CT synthesis with
no glioma downstream task; one-sided contrastive medical translation relies on
auxiliary segmentation supervision~\cite{seibel2024oct}. \todo{state the gap
explicitly: no within-modality MRI harmonizer keeps full cycle + multi-layer

\paragraph{Cycle-consistency (global anchor).}
For unpaired sets $\mathcal{D}_A,\mathcal{D}_B$ and generators
$G_{A\to B},G_{B\to A}$,
\begin{equation}
  \mathcal{L}_{\mathrm{cyc}} =
  \mathbb{E}_{x_a}\!\big[\|x_a-G_{B\to A}(G_{A\to B}(x_a))\|_1\big]
  +\mathbb{E}_{x_b}\!\big[\|x_b-G_{A\to B}(G_{B\to A}(x_b))\|_1\big].
  \label{eq:cyc}
\end{equation}

\paragraph{Multi-layer PatchNCE (local anchor).}
With encoder features at $L$ layers passed through 2-layer MLP heads $h_\ell$
and L2-normalized, for a query $z$, its positive $z^{+}$ at the corresponding
spatial location, and internal negatives $z^{-}$, the InfoNCE loss is
\begin{equation}
  \ell(z,z^{+},z^{-}) = -\log
  \frac{\exp(z\!\cdot\!z^{+}/\tau)}
       {\exp(z\!\cdot\!z^{+}/\tau)+\sum_{n}\exp(z\!\cdot\!z^{-}_{n}/\tau)},
  \qquad
  \mathcal{L}_{\mathrm{NCE}}=\mathbb{E}\sum_{\ell=1}^{L}\sum_{s}\ell(\cdot),
  \label{eq:nce}
\end{equation}
with temperature $\tau{=}0.07$, $L{=}5$ layers, and $256$ sampled locations
per layer (following CUT~\cite{park2020cut}). \todo{verify $\tau$, $L$, patch
count against the actual training config before camera-ready.}

\paragraph{Hybrid objective.}
PatchNCE \emph{augments} rather than replaces cycle-consistency:
\begin{equation}
  \mathcal{L} = \mathcal{L}_{\mathrm{adv}} + \mathcal{L}_{\mathrm{cyc}}
  + \mathcal{L}_{\mathrm{idt}} + \mathcal{L}_{\mathrm{ssim}}
  + \lambda_{\mathrm{NCE}}\,\mathcal{L}_{\mathrm{NCE}}.
  \label{eq:total}
\end{equation}
We sweep $\lambda_{\mathrm{NCE}}\in\{0.1,0.5,1.0,2.0\}$ at otherwise identical
hyperparameters. \todo{1 sentence: a selected $\lambda_{\mathrm{NCE}}$ below
